# 论文助手 · 七板块连续工作流 —— 公网部署依赖
#
# 本机开发**不用**这个文件：工作区里已经 vendored 好这些库，run.cmd / run.ps1
# 会把它们追加进 PYTHONPATH。容器里没有那些目录，才由 pip 安装这一份。
#
# ---- 直接依赖 -------------------------------------------------------------
# 由 tools/check_requirements.py 静态核对（AST 抠 import）。
# **注意 import 名与 PyPI 包名不是一回事**（docx→python-docx、yaml→PyYAML、
# fitz/pymupdf→PyMuPDF、pptx→python-pptx），对账脚本里有一张映射表。
#
#   fastapi        uvicorn        httpx         starlette
#   python-docx    PyYAML         lxml          python-multipart
#   PyMuPDF        jieba          python-pptx   numpy

fastapi>=0.115,<1.0
uvicorn>=0.30,<1.0
httpx>=0.27,<1.0
python-docx>=1.1,<2.0
PyYAML>=6.0,<7.0
lxml>=5.0
python-multipart>=0.0.9

# 文件导入（PDF 取文本）。**漏了它容器里"导入 PDF"会直接失效**——
# 本机开发时 pymupdf 来自 vendored libs，所以本机测试全过、容器里才炸，
# 正是 tools/check_requirements.py 存在的意义（它按源码 import 对账）。
pymupdf>=1.24

# starlette：fastapi 的底层框架。虽然它是 fastapi 的传递依赖，
# 但 ``suite/agents/bridge.py`` **直接 import** 了它（要手工构造 Request 让 agent
# 复用既有路由 handler）。直接依赖就该显式声明——否则哪天 fastapi 换掉底层实现，
# 自己代码里 import starlette 的地方会无声崩掉。版本跟随 fastapi 的约束。
starlette>=0.40,<1.0

# jieba：中文分词，本地 RAG 知识库要用。
# 为什么必须显式声明：FTS5 默认按空白切词，一整句中文会变成一个 token、
# 任何查询都匹配不上，所以入库前必须自己分词。缺了它 RAG 会**静默降级**为
# 单字切分（不会报错，但召回与精度都会变差），所以宁可在这里钉死。
jieba>=0.42

# python-pptx：答辩板块导出 .pptx 要用。
# 缺了它不会报错——代码会走 HTML 演示稿那条路（见 suite/defense.py 的 pptx_available），
# 但用户拿不到 .pptx。显式声明是为了让容器里默认就有。
python-pptx>=0.6.23

# numpy：语义检索（向量相似度）要用。
# 它常被别的库顺带装进来，所以本机开发时不容易发现漏了——
# 是 tools/check_requirements.py 按源码 import 对账才抓出来的。
# 不声明的话，容器里"语义检索"会报 ModuleNotFoundError 而不是优雅降级。
numpy>=1.24

# ---- 显式钉住的传递依赖 ---------------------------------------------------
# 这一条是**踩坑之后补上的**，不是想当然：
#
# fastapi / starlette 会带进 anyio，而 vendored 的 anyio 4.15.1 在导入时执行
# ``from typing_extensions import sentinel``。开发机上 site-packages 里的
# typing_extensions 版本偏旧、没有这个符号，于是启动直接失败：
#
#     ImportError: cannot import name 'sentinel' from 'typing_extensions'
#
# 容器里 pip 本来也会把 typing_extensions 作为 AnyIO/pydantic 的依赖装上，
# 但**装到哪个版本取决于解析结果**。这里显式钉一个下界，避免瘦镜像上
# 撞到同一个旧版本。任何 ≥4.12 的版本都有 sentinel。
typing-extensions>=4.12
